lastRefreshTime) {\n  // Render badge\n}\n\n// Badge HIDES when:\nif (refreshing) {\n  // Don't render badge (show progress bar instead)\n}\n\nif (!lastRefreshTime) {\n  // Never refreshed before - no badge\n}\n\n// STATUS DETERMINATION:\nconst diffMs = nowTick - lastRefreshTime;\nconst diffHours = Math.floor(diffMs / 3600000);\nconst diffDays = Math.floor(diffHours / 24);\n\nif (diffDays >= 1) {\n  statusClass = 'refresh-outdated-red';      // 24h+\n  icon = '!';                                 // Red status\n} else if (diffHours >= 12) {\n  statusClass = 'refresh-outdated-yellow';   // 12-24h\n  icon = '!';                                 // Yellow status\n} else {\n  statusClass = '';                           // < 12h\n  icon = '✓';                                 // Green status\n  // Animation applies: .last-refresh-info:not(.refresh-outdated-yellow):not(.refresh-outdated-red)\n}\n\n// TIME FORMAT:\nif (diffSecs < 60) display = 'Just now';\nif (diffMins < 60) display = `${diffMins}m ago`;\nif (diffHours < 24) display = `${diffHours}h ago`;\nif (diffDays >= 1) display = `${diffDays}d ago`;\n\n// TOOLTIP:\nexactTime = new Date(lastRefreshTime).toLocaleString();\n// Example: \"1/15/2024, 2:30:45 PM\"\n```\n\n## CSS Animation Specificity\n\n```css\n/* Glow animation selector specificity
